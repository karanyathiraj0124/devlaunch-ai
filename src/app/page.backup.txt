
"use client";

import { useState } from "react";
import { calculateJobMatch } from "@/lib/matching";

type Job = {
  job_id?: string;
  title?: string;
  job_title?: string;
  company_name?: string;
  company?: string;
  location?: string;
  via?: string;
  description?: string;
  share_link?: string;
  apply_link?: string;
  job_apply_link?: string;
  extensions?: string[];
};

export default function Home() {
  const [role, setRole] = useState("Frontend Developer");
  const [location, setLocation] = useState("Bangalore");
  const [skills, setSkills] = useState(
    "React, JavaScript, TypeScript"
  );

  const [jobs, setJobs] = useState<Job[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [searched, setSearched] = useState(false);

  async function findJobs(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();

    setLoading(true);
    setError("");
    setSearched(true);
    setJobs([]);

    try {
      const params = new URLSearchParams({
        role: role.trim(),
        location: location.trim(),
      });

      const response = await fetch(`/api/jobs?${params.toString()}`);
      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          data.error || "Unable to fetch jobs. Please try again."
        );
      }

      if (!Array.isArray(data.jobs)) {
        throw new Error("The jobs API returned an unexpected response.");
      }

      setJobs(data.jobs);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Something went wrong while fetching jobs."
      );
    } finally {
      setLoading(false);
    }
  }

  const userSkills = skills
    .split(",")
    .map((skill) => skill.trim())
    .filter(Boolean);

  return (
    <main className="min-h-screen bg-[#07090d] text-white">
      {/* NAVBAR */}
      <nav className="sticky top-0 z-20 border-b border-white/10 bg-[#07090d]/95 backdrop-blur">
        <div className="mx-auto flex max-w-7xl items-center justify-between px-5 py-5">
          <a href="#" className="text-2xl font-bold tracking-tight">
            Dev<span className="text-cyan-400">Launch</span>
            <span className="ml-2 text-xs font-normal text-gray-500">
              AI
            </span>
          </a>

          <div className="hidden items-center gap-8 text-sm text-gray-400 md:flex">
            <a href="#jobs" className="hover:text-cyan-400">
              Find Jobs
            </a>
            <a href="#how-it-works" className="hover:text-cyan-400">
              How It Works
            </a>
            <a href="#about" className="hover:text-cyan-400">
              About
            </a>
          </div>

          <a
            href="#search"
            className="rounded-full border border-white/20 px-4 py-2 text-sm hover:border-cyan-400 hover:text-cyan-400"
          >
            My Profile
          </a>
        </div>
      </nav>

      {/* HERO */}
      <section className="px-5 pb-12 pt-16 text-center md:pt-24">
        <div className="mx-auto max-w-4xl">
          <span className="inline-block rounded-full border border-cyan-400/30 bg-cyan-400/10 px-4 py-2 text-xs font-medium text-cyan-300">
            AI-powered developer career launchpad
          </span>

          <h1 className="mt-7 text-4xl font-bold leading-tight tracking-tight md:text-6xl">
            Your first developer
            <br />
            <span className="text-cyan-400">job starts here.</span>
          </h1>

          <p className="mx-auto mt-6 max-w-2xl leading-7 text-gray-400">
            Discover current developer jobs, compare your skills with
            employer requirements, and find out what to learn next.
          </p>
        </div>
      </section>

      {/* SEARCH FORM */}
      <section id="search" className="mx-auto max-w-5xl px-5 pb-12">
        <form
          onSubmit={findJobs}
          className="rounded-2xl border border-white/10 bg-[#101216] p-5 md:p-7"
        >
          <div className="grid gap-5 md:grid-cols-3">
            <div>
              <label
                htmlFor="role"
                className="mb-2 block text-sm text-gray-400"
              >
                Job Role
              </label>
              <input
                id="role"
                value={role}
                onChange={(event) => setRole(event.target.value)}
                placeholder="Frontend Developer"
                required
                className="w-full rounded-lg border border-white/10 bg-[#080a0e] px-4 py-3 text-sm outline-none transition focus:border-cyan-400"
              />
            </div>

            <div>
              <label
                htmlFor="location"
                className="mb-2 block text-sm text-gray-400"
              >
                Location
              </label>
              <input
                id="location"
                value={location}
                onChange={(event) => setLocation(event.target.value)}
                placeholder="Bangalore"
                required
                className="w-full rounded-lg border border-white/10 bg-[#080a0e] px-4 py-3 text-sm outline-none transition focus:border-cyan-400"
              />
            </div>

            <div>
              <label
                htmlFor="skills"
                className="mb-2 block text-sm text-gray-400"
              >
                Your Skills
              </label>
              <input
                id="skills"
                value={skills}
                onChange={(event) => setSkills(event.target.value)}
                placeholder="React, Python, SQL"
                className="w-full rounded-lg border border-white/10 bg-[#080a0e] px-4 py-3 text-sm outline-none transition focus:border-cyan-400"
              />
            </div>
          </div>

          <button
            type="submit"
            disabled={loading}
            className="mt-5 w-full rounded-lg bg-cyan-400 px-5 py-4 font-semibold text-black transition hover:bg-cyan-300 disabled:cursor-not-allowed disabled:opacity-60"
          >
            {loading ? "Searching for jobs..." : "⌕  Find My Jobs"}
          </button>

          <p className="mt-3 text-xs text-gray-500">
            Separate your skills with commas. Example: React, Python, SQL.
          </p>
        </form>
      </section>

      {/* JOB RESULTS */}
      <section id="jobs" className="mx-auto max-w-7xl px-5 pb-20">
        <div className="mb-7">
          <p className="text-xs font-semibold uppercase tracking-widest text-cyan-400">
            Live job results
          </p>

          <h2 className="mt-2 text-2xl font-bold">
            {searched ? "Jobs matching your search" : "Explore opportunities"}
          </h2>

          <p className="mt-2 text-sm text-gray-400">
            {loading
              ? "Fetching current job listings..."
              : searched
                ? `Found ${jobs.length} jobs for ${role || "your role"} in ${location || "your location"}.`
                : "Search for a role to discover developer opportunities."}
          </p>
        </div>

        {error && (
          <div
            role="alert"
            className="mb-6 rounded-xl border border-red-500/30 bg-red-500/10 p-4 text-sm text-red-300"
          >
            <p className="font-semibold">Unable to load jobs</p>
            <p className="mt-1">{error}</p>
            <button
              onClick={() =>
                document.getElementById("search")?.scrollIntoView({
                  behavior: "smooth",
                })
              }
              className="mt-3 underline"
            >
              Check your search and try again
            </button>
          </div>
        )}

        {loading && (
          <div className="grid gap-5 md:grid-cols-2 lg:grid-cols-3">
            {[1, 2, 3].map((item) => (
              <div
                key={item}
                className="animate-pulse rounded-2xl border border-white/10 bg-[#101216] p-6"
              >
                <div className="h-5 w-2/3 rounded bg-white/10" />
                <div className="mt-4 h-3 w-1/2 rounded bg-white/10" />
                <div className="mt-8 h-3 rounded bg-white/10" />
                <div className="mt-3 h-3 w-4/5 rounded bg-white/10" />
                <div className="mt-8 h-10 rounded bg-white/10" />
              </div>
            ))}
          </div>
        )}

        {!loading && searched && jobs.length === 0 && !error && (
          <div className="rounded-2xl border border-white/10 bg-[#101216] px-6 py-12 text-center">
            <h3 className="text-lg font-semibold">No jobs found</h3>
            <p className="mt-2 text-sm text-gray-400">
              Try another job title or location.
            </p>
          </div>
        )}

        {!loading && jobs.length > 0 && (
          <div className="grid gap-5 md:grid-cols-2 lg:grid-cols-3">
            {jobs.map((job, index) => {
              const title =
                job.title || job.job_title || "Developer Opportunity";

              const company =
                job.company_name || job.company || "Company not listed";

              const description = job.description || "";

              const match = calculateJobMatch(
                userSkills,
                `${title} ${description}`
              );

              const applyUrl =
                job.apply_link ||
                job.job_apply_link ||
                job.share_link ||
                "";

              const jobKey =
                job.job_id || `${title}-${company}-${index}`;

              return (
                <article
                  key={jobKey}
                  className="flex flex-col rounded-2xl border border-white/10 bg-[#101216] p-5 transition duration-200 hover:-translate-y-1 hover:border-cyan-400/40"
                >
                  <div className="flex items-start justify-between gap-3">
                    <h3 className="text-lg font-bold leading-6">
                      {title}
                    </h3>

                    <span className="shrink-0 rounded-full bg-cyan-400/10 px-2 py-1 text-[10px] font-semibold uppercase text-cyan-300">
                      Live
                    </span>
                  </div>

                  <p className="mt-2 font-medium text-cyan-400">
                    {company}
                  </p>

                  <p className="mt-3 text-sm text-gray-400">
                    📍 {job.location || location}
                  </p>

                  {job.via && (
                    <p className="mt-2 text-xs text-gray-500">
                      Via {job.via}
                    </p>
                  )}

                  {job.extensions && job.extensions.length > 0 && (
                    <div className="mt-3 flex flex-wrap gap-2">
                      {job.extensions.slice(0, 3).map((extension, i) => (
                        <span
                          key={`${extension}-${i}`}
                          className="rounded-full border border-white/10 px-2 py-1 text-[10px] text-gray-300"
                        >
                          {extension}
                        </span>
                      ))}
                    </div>
                  )}

                  {/* MATCH SCORE */}
                  <div className="mt-5 rounded-xl border border-cyan-400/20 bg-cyan-400/5 p-4">
                    <div className="flex items-center justify-between gap-2">
                      <span className="text-sm text-gray-300">
                        Skill overlap
                      </span>
                      <span className="text-xl font-bold text-cyan-400">
                        {match.score}%
                      </span>
                    </div>

                    <div
                      className="mt-3 h-2 overflow-hidden rounded-full bg-white/10"
                      role="progressbar"
                      aria-label="Skill overlap percentage"
                      aria-valuenow={match.score}
                      aria-valuemin={0}
                      aria-valuemax={100}
                    >
                      <div
                        className="h-full rounded-full bg-cyan-400 transition-all"
                        style={{ width: `${match.score}%` }}
                      />
                    </div>

                    <p className="mt-3 text-xs leading-5 text-gray-400">
                      <span className="text-gray-200">Skills found: </span>
                      {match.matchedSkills.join(", ") ||
                        "No listed skills identified in the description."}
                    </p>

                    <p className="mt-2 text-xs leading-5 text-gray-400">
                      <span className="text-gray-200">
                        Skills to check:{" "}
                      </span>
                      {match.missingSkills.join(", ") ||
                        "All your listed skills were found."}
                    </p>
                  </div>

                  <p className="mt-4 flex-1 text-sm leading-6 text-gray-400">
                    {description
                      ? `${description.slice(0, 230)}${description.length > 230 ? "..." : ""}`
                      : "Open the job listing to view the full description and requirements."}
                  </p>

                  {applyUrl ? (
                    <a
                      href={applyUrl}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="mt-5 block rounded-lg bg-white px-4 py-3 text-center text-sm font-semibold text-black transition hover:bg-cyan-300"
                    >
                      View Job & Apply ↗
                    </a>
                  ) : (
                    <p className="mt-5 rounded-lg border border-white/10 px-4 py-3 text-center text-sm text-gray-500">
                      Application link unavailable
                    </p>
                  )}
                </article>
              );
            })}
          </div>
        )}
      </section>

      {/* HOW IT WORKS */}
      <section
        id="how-it-works"
        className="border-t border-white/10 bg-[#0b0d12] px-5 py-20"
      >
        <div className="mx-auto max-w-7xl">
          <div className="text-center">
            <p className="text-xs font-semibold uppercase tracking-widest text-cyan-400">
              How it works
            </p>
            <h2 className="mt-3 text-3xl font-bold">
              From searching to job-ready
            </h2>
          </div>

          <div className="mt-12 grid gap-5 md:grid-cols-3">
            <Feature
              number="01"
              title="Find Current Jobs"
              description="Search developer opportunities using live job-search results."
            />
            <Feature
              number="02"
              title="Check Your Match"
              description="Compare your listed skills with skills mentioned in job descriptions."
            />
            <Feature
              number="03"
              title="Plan Your Growth"
              description="Use identified skill gaps as a starting point for your learning plan."
            />
          </div>
        </div>
      </section>

      {/* ABOUT */}
      <footer
        id="about"
        className="border-t border-white/10 px-5 py-8 text-center"
      >
        <p className="font-bold">
          Dev<span className="text-cyan-400">Launch</span> AI
        </p>
        <p className="mt-2 text-sm text-gray-500">
          Helping developers discover opportunities and plan their next steps.
        </p>
        <p className="mt-4 text-xs text-gray-600">
          Job listings are provided by the connected search API. Always verify
          requirements and application details on the employer&apos;s website.
        </p>
      </footer>
    </main>
  );
}

function Feature({
  number,
  title,
  description,
}: {
  number: string;
  title: string;
  description: string;
}) {
  return (
    <div className="rounded-2xl border border-white/10 bg-[#101216] p-6">
      <span className="text-sm font-bold text-cyan-400">{number}</span>
      <h3 className="mt-4 text-lg font-semibold">{title}</h3>
      <p className="mt-3 text-sm leading-6 text-gray-400">{description}</p>
    </div>
  );
}
